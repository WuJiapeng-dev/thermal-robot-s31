- `thermal_robot\stm32\inc`
- SPL：`STM32F10x_StdPeriph_Driver\inc`
- CMSIS core：`Libraries\CMSIS\CM3\CoreSupport`
- CMSIS device：`Libraries\CMSIS\CM3\DeviceSupport\ST\STM32F10x`

## 6. 编译与链接设置

- Target：STM32F103CB（128KB Flash / 20KB RAM）
- Crystal：8.000MHz
- ARMCC：默认 C99；勾选 SPL 要求的 `USE_STDPERIPH_DRIVER`、`STM32F10X_MD`
- Linker：Keil 自动选择 `STM32F10x_MD` 散列文件（128KB Flash / 20KB RAM）
- Debug：ST-Link，SWD（PA13/PA14）。代码保留 SWD、关闭 JTAG，禁止改回 JTAG。

## 7. 上电验证顺序

1. 烧录后先运行 USART1 调试台（115200）：输入 `help`。
2. 输入 `lcdtest`：ILI9341 应显示彩条、ASCII 与中文降级提示。
3. 输入 `bat`：读取 PA4 ADC 分压电池电压。
4. 接好 S31 的 USART2 交叉线后，观察界面刷新；输入 `link` 查看最近 5 帧 JSON。
5. `mtest/etest/go/turn/scan` 当前应返回“本功能未启用（采购清单无驱动板）”；补件后将 `board_config.h` 的 `ENABLE_MOTION_DRIVE` 改为 `1`。

## 8. 当前字库状态

`font12.c` 已编码 72 个界面高频字的 GB2312/Unicode 目录，但本环境没有可信的 12×12 字模来源；所有 bitmap 均为 `0`，缺字按 `?` 降级，不会花屏。取得可信字模后填入 24 字节/字（逐行、高字节在前、每行低 4 位无效）。

## 9. 启动文件里必须改的一个数值

`startup_stm32f10x_md.s` 默认 `Stack_Size EQU 0x00000400`（1KB）。本工程的栈上会放一个 `json_frame_t`（约 144 字节）加若干行缓冲与浮点运算临时量，1KB 偏紧，建议改成：

```asm
Stack_Size      EQU     0x00000800      ; 2KB
Heap_Size       EQU     0x00000200      ; 0x200，保持默认
```

说明：本工程代码里没有 `malloc/free`，RAM 全部用静态缓冲；但 `debug_console.c` 的 `printf/snprintf` 在 ARMCC 标准库 + 非 microlib 下可能仍会用到少量堆，因此 `Heap_Size` 建议保持默认 `0x200`，不要归零。若确认改用 microlib，可再省这 512 字节。

#ifndef JSON_FRAME_H
#define JSON_FRAME_H

#include <stdint.h>
#include "board_config.h"

typedef enum {
    JSON_TYPE_NONE = 0,
    JSON_TYPE_HOT,
    JSON_TYPE_HB,
    JSON_TYPE_ST
} json_message_type_t;

typedef struct {
    json_message_type_t type;
    int32_t id;
    char grp[JSON_GROUP_CHARS];
    char zh[JSON_ZH_CHARS];
    int32_t x;
    int32_t y;
    float t;
    float dt;
    int32_t px;
    float conf;
    uint32_t ts;
    int32_t n;
    char stage[JSON_STAGE_CHARS];
    int32_t prog;
    uint32_t missing_mask;
} json_frame_t;

enum {
    JSON_MISSING_TYPE = 1UL << 0,
    JSON_MISSING_ID   = 1UL << 1,
    JSON_MISSING_TEMP = 1UL << 2,
    JSON_MISSING_GRP  = 1UL << 3,
    JSON_MISSING_ZH   = 1UL << 4,
    JSON_MISSING_X    = 1UL << 5,
    JSON_MISSING_Y    = 1UL << 6,
    JSON_MISSING_DT   = 1UL << 7,
    JSON_MISSING_PX   = 1UL << 8,
    JSON_MISSING_CONF = 1UL << 9,
    JSON_MISSING_TS   = 1UL << 10,
    JSON_MISSING_N    = 1UL << 11,
    JSON_MISSING_STAGE = 1UL << 12,
    JSON_MISSING_PROG = 1UL << 13
};

uint8_t json_parse_frame(const char *text, uint32_t length, json_frame_t *out);

#endif

#include "board_config.h"
#include <stdio.h>

#define COMM_NO_FRAME 0xFFU

/* 每块 512B DMA 数据 + 1B 终止符；两块交替，禁止逐字节中断。 */
static uint8_t g_rx_buffer[COMM_RX_BUFFER_COUNT][COMM_RX_FRAME_MAX + 1U];
static volatile uint8_t g_active_buffer;
static volatile uint8_t g_ready_buffer = COMM_NO_FRAME;
static volatile uint16_t g_ready_length;
static volatile uint32_t g_oversize_count;
static volatile uint32_t g_drop_count;

static void comm_rearm(uint8_t index)
{
    DMA_SetCurrDataCounter(COMM_DMA_CHANNEL, COMM_RX_FRAME_MAX);
    DMA_ClearFlag(DMA1_FLAG_TC6 | DMA1_FLAG_HT6 | DMA1_FLAG_TE6);
    g_active_buffer = index;
    DMA_Cmd(COMM_DMA_CHANNEL, ENABLE);
}

void comm_init(void)
{

    RCC_APB1PeriphClockCmd(COMM_USART_CLK, ENABLE);

    usart.USART_Parity = USART_Parity_No;
    usart.USART_HardwareFlowControl = USART_HardwareFlowControl_None;
    usart.USART_Mode = USART_Mode_Rx | USART_Mode_Tx;
    USART_Init(COMM_USART, &usart);

    DMA_DeInit(COMM_DMA_CHANNEL);
    DMA_StructInit(&dma);
    dma.DMA_PeripheralBaseAddr = (uint32_t)&COMM_USART->DR;
    dma.DMA_MemoryBaseAddr = (uint32_t)&g_rx_buffer[0][0];
    dma.DMA_DIR = DMA_DIR_PeripheralSRC;
    dma.DMA_BufferSize = COMM_RX_FRAME_MAX;
    dma.DMA_PeripheralInc = DMA_PeripheralInc_Disable;

    dma.DMA_MemoryDataSize = DMA_MemoryDataSize_Byte;
    dma.DMA_Mode = DMA_Mode_Normal;
    dma.DMA_Priority = DMA_Priority_High;
    dma.DMA_M2M = DMA_M2M_Disable;
    DMA_Init(COMM_DMA_CHANNEL, &dma);

    nvic.NVIC_IRQChannel = COMM_USART_IRQn;
    nvic.NVIC_IRQChannelPreemptionPriority = 1U;
    nvic.NVIC_IRQChannelSubPriority = 1U;
    nvic.NVIC_IRQChannelCmd = ENABLE;

void comm_usart2_irq(void)
{
    volatile uint32_t dummy;
    uint32_t status;
    uint16_t remaining;
    uint16_t length;
    uint8_t completed;
    uint8_t overrun;

    if (USART_GetITStatus(COMM_USART, USART_IT_IDLE) == RESET) { return; }
    status = COMM_USART->SR;
    overrun = ((status & USART_SR_ORE) != 0U) ? 1U : 0U;
    dummy = status;
    dummy = COMM_USART->DR;             /* SR 后读 DR，清 IDLE/ORE；不打印、不延时。 */
    (void)dummy;

    DMA_Cmd(COMM_DMA_CHANNEL, DISABLE);
    remaining = DMA_GetCurrDataCounter(COMM_DMA_CHANNEL);
    length = (uint16_t)(COMM_RX_FRAME_MAX - remaining);
    completed = g_active_buffer;
    if (length > COMM_RX_FRAME_MAX) { length = COMM_RX_FRAME_MAX; }

    if (length == 0U) {
        comm_rearm(completed);
        return;
    }
    if (overrun || (length == COMM_RX_FRAME_MAX && remaining != 0U)) {
        g_rx_buffer[completed][0] = 0U;
        if (length == COMM_RX_FRAME_MAX) { ++g_oversize_count; }
        else { ++g_drop_count; }
        comm_rearm(completed);
        return;
    }

    g_rx_buffer[completed][length] = 0U;
    if (g_ready_buffer == COMM_NO_FRAME) {
        g_ready_length = length;
        g_ready_buffer = completed;
        comm_rearm((uint8_t)(completed ^ 1U));
    } else {
        /* 前一帧尚未由主循环释放，丢弃新帧而不是覆盖正在解析的数据。 */
        ++g_drop_count;
        comm_rearm(completed);
    }
}

uint8_t comm_frame_available(void)
{
    return (g_ready_buffer != COMM_NO_FRAME) ? 1U : 0U;
}

const char *comm_acquire_frame(uint16_t *length)
{
    if (g_ready_buffer == COMM_NO_FRAME || length == 0) { return 0; }
    *length = g_ready_length;
    return (const char *)&g_rx_buffer[g_ready_buffer][0];
}

void comm_release_frame(void)

    __DMB();
    g_ready_buffer = COMM_NO_FRAME;
}

uint32_t comm_oversize_count(void)

    return g_oversize_count;
}

{
    return g_drop_count;
}

    comm_write(frame);
}

#include "json_frame.h"
#include <string.h>

#define JSON_MAX_DEPTH 8U

typedef struct {
    const char *text;
    uint32_t length;
    uint32_t pos;
} json_cursor_t;

static void skip_space(json_cursor_t *c)
{
    while (c->pos < c->length) {
        char ch = c->text[c->pos];
        if (ch != ' ' && ch != '\t' && ch != '\r' && ch != '\n') { break; }
        ++c->pos;
    }
}

static uint8_t take(json_cursor_t *c, char expected)
{
    skip_space(c);
    if (c->pos >= c->length || c->text[c->pos] != expected) { return 0U; }
    ++c->pos;
    return 1U;
}

static uint8_t hex_digit(char ch, uint8_t *value)
{
    if (ch >= '0' && ch <= '9') { *value = (uint8_t)(ch - '0'); return 1U; }
    if (ch >= 'a' && ch <= 'f') { *value = (uint8_t)(ch - 'a' + 10); return 1U; }
    if (ch >= 'A' && ch <= 'F') { *value = (uint8_t)(ch - 'A' + 10); return 1U; }
    return 0U;
}

static void emit_byte(char *out, uint32_t size, uint32_t *used, char ch, uint8_t *ok)
{
    if (out != 0) {
        if (*used + 1U >= size) { *ok = 0U; return; }
        out[*used] = ch;
    }
    ++*used;
}

static uint8_t parse_hex4(json_cursor_t *c, uint32_t *value)
{
    uint8_t i;
    uint8_t digit;
    *value = 0U;
    if (c->length - c->pos < 4U) { return 0U; }
    for (i = 0U; i < 4U; ++i) {
        if (!hex_digit(c->text[c->pos++], &digit)) { return 0U; }
        *value = (*value << 4) | digit;
    }
    return 1U;
}

static void emit_utf8(uint32_t cp, char *out, uint32_t size, uint32_t *used, uint8_t *ok)
{
    if (cp <= 0x7FU) {
        emit_byte(out, size, used, (char)cp, ok);
    } else if (cp <= 0x7FFU) {
        emit_byte(out, size, used, (char)(0xC0U | (cp >> 6)), ok);
        emit_byte(out, size, used, (char)(0x80U | (cp & 0x3FU)), ok);
    } else {
        emit_byte(out, size, used, (char)(0xE0U | (cp >> 12)), ok);
        emit_byte(out, size, used, (char)(0x80U | ((cp >> 6) & 0x3FU)), ok);
        emit_byte(out, size, used, (char)(0x80U | (cp & 0x3FU)), ok);
    }
}

static uint8_t parse_string(json_cursor_t *c, char *out, uint32_t size)
{
    uint32_t used = 0U;
    uint8_t ok = 1U;
    skip_space(c);
    if (c->pos >= c->length || c->text[c->pos++] != '"') { return 0U; }
    while (c->pos < c->length) {
        uint8_t ch = (uint8_t)c->text[c->pos++];
        if (ch == '"') {
            if (out != 0 && size > 0U) {
                if (used >= size) { return 0U; }
                out[used] = '\0';
            }
            return ok;
        }
        if (ch < 0x20U) { return 0U; }
        if (ch == '\\') {
            uint32_t cp;
            if (c->pos >= c->length) { return 0U; }
            ch = (uint8_t)c->text[c->pos++];
            switch (ch) {
            case '"': case '\\': case '/': emit_byte(out, size, &used, (char)ch, &ok); break;
            case 'b': emit_byte(out, size, &used, '\b', &ok); break;
            case 'f': emit_byte(out, size, &used, '\f', &ok); break;
            case 'n': emit_byte(out, size, &used, '\n', &ok); break;
            case 'r': emit_byte(out, size, &used, '\r', &ok); break;
            case 't': emit_byte(out, size, &used, '\t', &ok); break;
            case 'u':
                if (!parse_hex4(c, &cp) || (cp >= 0xD800U && cp <= 0xDFFFU)) { return 0U; }
                emit_utf8(cp, out, size, &used, &ok);
                break;
            default: return 0U;
            }
        } else {
            emit_byte(out, size, &used, (char)ch, &ok);
        }
    }
    return 0U;
}

static uint8_t parse_number(json_cursor_t *c, float *value)
{
    int sign = 1;
    int exponent_sign = 1;
    int exponent = 0;
    uint8_t digits = 0U;
    float number = 0.0f;
    float scale = 0.1f;
    skip_space(c);
    if (c->pos < c->length && c->text[c->pos] == '-') { sign = -1; ++c->pos; }
    while (c->pos < c->length && c->text[c->pos] >= '0' && c->text[c->pos] <= '9') {
        number = number * 10.0f + (float)(c->text[c->pos++] - '0');
        digits = 1U;
    }
    if (c->pos < c->length && c->text[c->pos] == '.') {
        ++c->pos;
        while (c->pos < c->length && c->text[c->pos] >= '0' && c->text[c->pos] <= '9') {
            number += (float)(c->text[c->pos++] - '0') * scale;
            scale *= 0.1f;
            digits = 1U;
        }
    }
    if (!digits) { return 0U; }
    if (c->pos < c->length && (c->text[c->pos] == 'e' || c->text[c->pos] == 'E')) {
        ++c->pos;
        if (c->pos < c->length && (c->text[c->pos] == '+' || c->text[c->pos] == '-')) {
            if (c->text[c->pos++] == '-') { exponent_sign = -1; }
        }
        digits = 0U;
        while (c->pos < c->length && c->text[c->pos] >= '0' && c->text[c->pos] <= '9') {
            if (exponent < 38) { exponent = exponent * 10 + (c->text[c->pos] - '0'); }
            ++c->pos;
            digits = 1U;
        }
        if (!digits) { return 0U; }
        while (exponent-- > 0) {
            number = (exponent_sign > 0) ? number * 10.0f : number * 0.1f;
        }
    }
    *value = sign > 0 ? number : -number;
    return 1U;
}

static uint8_t skip_value(json_cursor_t *c, uint8_t depth)
{
    char ch;
    if (depth > JSON_MAX_DEPTH) { return 0U; }
    skip_space(c);
    if (c->pos >= c->length) { return 0U; }
    ch = c->text[c->pos];
    if (ch == '"') { return parse_string(c, 0, 0U); }
    if (ch == '{' || ch == '[') {
        char close = ch == '{' ? '}' : ']';
        ++c->pos;
        skip_space(c);
        if (c->pos < c->length && c->text[c->pos] == close) { ++c->pos; return 1U; }
        for (;;) {
            if (ch == '{' && !parse_string(c, 0, 0U)) { return 0U; }
            if (ch == '{' && !take(c, ':')) { return 0U; }
            if (!skip_value(c, (uint8_t)(depth + 1U))) { return 0U; }
            skip_space(c);
            if (c->pos < c->length && c->text[c->pos] == ',') { ++c->pos; continue; }
            if (c->pos < c->length && c->text[c->pos] == close) { ++c->pos; return 1U; }
            return 0U;
        }
    }
    if (ch == '-' || (ch >= '0' && ch <= '9')) {
        float ignored;
        return parse_number(c, &ignored);
    }
    if (c->length - c->pos >= 4U && memcmp(&c->text[c->pos], "true", 4U) == 0) {
        c->pos += 4U; return 1U;
    }
    if (c->length - c->pos >= 5U && memcmp(&c->text[c->pos], "false", 5U) == 0) {
        c->pos += 5U; return 1U;
    }
    if (c->length - c->pos >= 4U && memcmp(&c->text[c->pos], "null", 4U) == 0) {
        c->pos += 4U; return 1U;
    }
    return 0U;
}

uint8_t json_parse_frame(const char *text, uint32_t length, json_frame_t *out)
{
    json_cursor_t c;
    char type[8] = "";
    uint32_t seen = 0U;
    uint8_t ok = 1U;

    if (text == 0 || out == 0 || length == 0U) { return 0U; }
    memset(out, 0, sizeof(*out));
    out->id = -1;
    out->t = TEMPERATURE_INVALID;
    (void)strcpy(out->grp, "--");
    (void)strcpy(out->zh, "--");
    (void)strcpy(out->stage, "--");
    c.text = text; c.length = length; c.pos = 0U;
    if (!take(&c, '{')) { return 0U; }
    skip_space(&c);
    if (c.pos < c.length && c.text[c.pos] == '}') { return 0U; }

    while (ok && c.pos < c.length) {
        char key[16];
        float number;
        if (!parse_string(&c, key, sizeof(key)) || !take(&c, ':')) { return 0U; }
        if (strcmp(key, "type") == 0) {
            ok = parse_string(&c, type, sizeof(type)); seen |= JSON_MISSING_TYPE;
        } else if (strcmp(key, "grp") == 0) {
            ok = parse_string(&c, out->grp, sizeof(out->grp)); seen |= JSON_MISSING_GRP;
        } else if (strcmp(key, "zh") == 0) {
            ok = parse_string(&c, out->zh, sizeof(out->zh)); seen |= JSON_MISSING_ZH;
        } else if (strcmp(key, "stage") == 0) {
            ok = parse_string(&c, out->stage, sizeof(out->stage)); seen |= JSON_MISSING_STAGE;
        } else if (strcmp(key, "id") == 0 || strcmp(key, "x") == 0 || strcmp(key, "y") == 0 ||
                   strcmp(key, "t") == 0 || strcmp(key, "dt") == 0 || strcmp(key, "px") == 0 ||
                   strcmp(key, "conf") == 0 || strcmp(key, "ts") == 0 || strcmp(key, "n") == 0 ||
                   strcmp(key, "prog") == 0) {
            if (!parse_number(&c, &number)) { return 0U; }
            if (strcmp(key, "id") == 0) { out->id = (int32_t)number; seen |= JSON_MISSING_ID; }
            else if (strcmp(key, "x") == 0) { out->x = (int32_t)number; seen |= JSON_MISSING_X; }
            else if (strcmp(key, "y") == 0) { out->y = (int32_t)number; seen |= JSON_MISSING_Y; }
            else if (strcmp(key, "t") == 0) { out->t = number; seen |= JSON_MISSING_TEMP; }
            else if (strcmp(key, "dt") == 0) { out->dt = number; seen |= JSON_MISSING_DT; }
            else if (strcmp(key, "px") == 0) { out->px = (int32_t)number; seen |= JSON_MISSING_PX; }
            else if (strcmp(key, "conf") == 0) { out->conf = number; seen |= JSON_MISSING_CONF; }
            else if (strcmp(key, "ts") == 0) { out->ts = (uint32_t)number; seen |= JSON_MISSING_TS; }
            else if (strcmp(key, "n") == 0) { out->n = (int32_t)number; seen |= JSON_MISSING_N; }
            else { out->prog = (int32_t)number; seen |= JSON_MISSING_PROG; }
        } else {
            ok = skip_value(&c, 0U);
        }
        if (!ok) { return 0U; }
        skip_space(&c);
        if (c.pos < c.length && c.text[c.pos] == ',') { ++c.pos; continue; }
        if (c.pos < c.length && c.text[c.pos] == '}') { ++c.pos; break; }
        return 0U;
    }
    skip_space(&c);
    if (c.pos != c.length) { return 0U; }

    if ((seen & JSON_MISSING_TYPE) == 0U) { return 0U; }
    if (strcmp(type, "hot") == 0) {
        out->type = JSON_TYPE_HOT;
        out->missing_mask = (uint32_t)(~seen) & 0x03FEUL;
    } else if (strcmp(type, "hb") == 0) {
        out->type = JSON_TYPE_HB;
        out->missing_mask = (uint32_t)(~seen) & (JSON_MISSING_TS | JSON_MISSING_N);
    } else if (strcmp(type, "st") == 0) {
        out->type = JSON_TYPE_ST;
        out->missing_mask = (uint32_t)(~seen) & (JSON_MISSING_STAGE | JSON_MISSING_PROG);
    } else {
        return 0U;
    }
    return 1U;
}

{
    uint32_t timeout = LCD_SPI_TIMEOUT;
    while (SPI_I2S_GetFlagStatus(LCD_SPI, SPI_I2S_FLAG_TXE) == RESET) {
        if (--timeout == 0U) { return 0U; }
    }
    SPI_I2S_SendData(LCD_SPI, value);
    timeout = LCD_SPI_TIMEOUT;
    while (SPI_I2S_GetFlagStatus(LCD_SPI, SPI_I2S_FLAG_BSY) == SET) {
        if (--timeout == 0U) { return 0U; }
    }
    return 1U;
}

void lcd_fill_rect(uint16_t x, uint16_t y, uint16_t w, uint16_t h, uint16_t color)
{

    uint16_t i;
    uint8_t hi = (uint8_t)(color >> 8);
    uint8_t lo = (uint8_t)color;

    if ((x >= LCD_WIDTH) || (y >= LCD_HEIGHT) || (w == 0U) || (h == 0U)) { return; }
    if ((uint32_t)x + w > LCD_WIDTH) { w = LCD_WIDTH - x; }
    if ((uint32_t)y + h > LCD_HEIGHT) { h = LCD_HEIGHT - y; }
    for (i = 0U; i < w; ++i) {
        g_lcd_line[i * 2U] = hi;
        g_lcd_line[i * 2U + 1U] = lo;
    }
    lcd_set_window(x, y, (uint16_t)(x + w - 1U), (uint16_t)(y + h - 1U));
    lcd_select(1U);
    GPIO_SetBits(LCD_DC_PORT, LCD_DC_PIN);
    for (row = 0U; row < h; ++row) {
        for (i = 0U; i < (uint16_t)(w * 2U); ++i) {
            if (lcd_spi_byte(g_lcd_line[i]) == 0U) {
                lcd_select(0U);
                return;
            }
        }
    }
    lcd_select(0U);
}

static void lcd_draw_ascii(uint16_t x, uint16_t y, char ch, uint16_t fg,
                           uint16_t bg, uint8_t scale)
{

    uint8_t col;
    uint8_t index = ((uint8_t)ch >= 32U && (uint8_t)ch <= 127U) ?
                    ((uint8_t)ch - 32U) : (uint8_t)('?' - 32U);

    lcd_fill_rect(x, y, (uint16_t)(6U * scale), (uint16_t)(8U * scale), bg);

                lcd_fill_rect((uint16_t)(x + col * scale), (uint16_t)(y + row * scale),
                              scale, scale, fg);
            }
        }
    }
}

static uint32_t utf8_decode(const uint8_t *s, uint8_t *consumed)
{
    if (s[0] < 0x80U) { *consumed = 1U; return s[0]; }
    if ((s[0] & 0xE0U) == 0xC0U && (s[1] & 0xC0U) == 0x80U) {
        *consumed = 2U;
        return ((uint32_t)(s[0] & 0x1FU) << 6) | (uint32_t)(s[1] & 0x3FU);
    }
    if ((s[0] & 0xF0U) == 0xE0U && (s[1] & 0xC0U) == 0x80U &&
        (s[2] & 0xC0U) == 0x80U) {
        *consumed = 3U;
        return ((uint32_t)(s[0] & 0x0FU) << 12) |
               ((uint32_t)(s[1] & 0x3FU) << 6) | (uint32_t)(s[2] & 0x3FU);
    }
    if ((s[0] & 0xF8U) == 0xF0U && (s[1] & 0xC0U) == 0x80U &&
        (s[2] & 0xC0U) == 0x80U && (s[3] & 0xC0U) == 0x80U) {
        *consumed = 4U;
        return ((uint32_t)(s[0] & 0x07U) << 18) |
               ((uint32_t)(s[1] & 0x3FU) << 12) |
               ((uint32_t)(s[2] & 0x3FU) << 6) | (uint32_t)(s[3] & 0x3FU);
    }
    *consumed = 1U;
    return 0xFFFFFFFFUL;
}

static void lcd_draw_chinese(uint16_t x, uint16_t y, const uint8_t *glyph,
                             uint16_t fg, uint16_t bg, uint8_t scale)
{
    uint8_t row;
    uint8_t col;
    uint16_t bits;

    lcd_fill_rect(x, y, (uint16_t)(12U * scale), (uint16_t)(12U * scale), bg);
    if (glyph == 0) {
        lcd_draw_ascii((uint16_t)(x + 3U * scale), (uint16_t)(y + 2U * scale),
                       '?', fg, bg, scale);
        return;
    }
    for (row = 0U; row < 12U; ++row) {
        bits = (uint16_t)(((uint16_t)glyph[row * 2U] << 8) | glyph[row * 2U + 1U]);
        for (col = 0U; col < 12U; ++col) {
            if ((bits & (uint16_t)(0x8000U >> col)) != 0U) {
                lcd_fill_rect((uint16_t)(x + col * scale), (uint16_t)(y + row * scale),
                              scale, scale, fg);
            }
        }
    }
}

void lcd_draw_string(uint16_t x, uint16_t y, const char *text,
                     uint16_t fg, uint16_t bg, uint8_t scale)
{
    uint32_t cursor = x;
    const uint8_t *s = (const uint8_t *)text;
    uint8_t consumed;
    uint32_t codepoint;
    const uint8_t *glyph;

    if ((text == 0) || (scale == 0U)) { return; }
    while ((*s != 0U) && (cursor < LCD_WIDTH)) {
        codepoint = utf8_decode(s, &consumed);
        s += consumed;
        if (codepoint < 0x80U) {
            lcd_draw_ascii((uint16_t)cursor, y, (char)codepoint, fg, bg, scale);
            cursor += 6U * scale;
        } else {
            glyph = (codepoint == 0xFFFFFFFFUL) ? 0 : font12_get_unicode(codepoint);
            lcd_draw_chinese((uint16_t)cursor, y, glyph, fg, bg, scale);
            cursor += 12U * scale;
        }
    }
}

    uint16_t samples[BATTERY_MEDIAN_SAMPLES];
    uint8_t i;
    uint8_t j;

    for (i = 0U; i < BATTERY_MEDIAN_SAMPLES; ++i) {
        samples[i] = adc_read_once();
    }
    for (i = 0U; i < BATTERY_MEDIAN_SAMPLES; ++i) {
        for (j = (uint8_t)(i + 1U); j < BATTERY_MEDIAN_SAMPLES; ++j) {
            if (samples[j] < samples[i]) {
                uint16_t temp = samples[i];

/* TB6612 两路输入真值表（方案说明书 5.5）：正向说明见下。
 *   IN1=1, IN2=0  -> 正转（PWM 调速）
 *   IN1=0, IN2=1  -> 反转
 *   IN1=1, IN2=1  -> 短路制动（PWM 无效）
 *   IN1=0, IN2=0  -> 停止 / 滑行（输出高阻）
 *
 * 早期版本用 `sign >= 0` / `sign <= 0` 从同一个 sign 推导两根脚，导致：
 *   sign = 0 -> 输出 (1,1) 即制动，而不是滑行；
 *   sign = +1 -> 输出 (1,0) 即正转；
 * 于是 motor_brake() 实际输出「正转 + 满占空比」，motor_coast() 实际输出制动，
 * 两个语义完全互换。上电后第一次调用 motor_brake() 会让车体全速前冲。
 * 因此这里改为显式方向枚举，不再从数值符号推导。 */
typedef enum {
    MOTOR_DIR_COAST = 0,
    MOTOR_DIR_FORWARD,
    MOTOR_DIR_REVERSE,
    MOTOR_DIR_BRAKE
} motor_dir_t;

static int16_t g_left;
static int16_t g_right;
static int8_t g_left_sign;
static int8_t g_right_sign;
static uint8_t g_initialized;

static void write_channel(GPIO_TypeDef *in1_port, uint16_t in1_pin,
                          GPIO_TypeDef *in2_port, uint16_t in2_pin,
                          motor_dir_t dir)
{
    switch (dir) {
    case MOTOR_DIR_FORWARD:
        GPIO_WriteBit(in1_port, in1_pin, Bit_SET);
        GPIO_WriteBit(in2_port, in2_pin, Bit_RESET);
        break;
    case MOTOR_DIR_REVERSE:
        GPIO_WriteBit(in1_port, in1_pin, Bit_RESET);
        GPIO_WriteBit(in2_port, in2_pin, Bit_SET);
        break;
    case MOTOR_DIR_BRAKE:
        GPIO_WriteBit(in1_port, in1_pin, Bit_SET);
        GPIO_WriteBit(in2_port, in2_pin, Bit_SET);
        break;
    default: /* MOTOR_DIR_COAST */
        GPIO_WriteBit(in1_port, in1_pin, Bit_RESET);
        GPIO_WriteBit(in2_port, in2_pin, Bit_RESET);
        break;
    }
}

static motor_dir_t dir_from_speed(int16_t value)
{
    if (value > 0) {
        return MOTOR_DIR_FORWARD;
    }
    if (value < 0) {
        return MOTOR_DIR_REVERSE;
    }
    return MOTOR_DIR_COAST;
}

static void write_directions(motor_dir_t left_dir, motor_dir_t right_dir)
{
    /* VM 仅 7.4~8.4V，严禁 12V；STBY 硬件接 3.3V。 */
    write_channel(MOTOR_AIN1_PORT, MOTOR_AIN1_PIN,
                  MOTOR_AIN2_PORT, MOTOR_AIN2_PIN, left_dir);
    write_channel(MOTOR_BIN1_PORT, MOTOR_BIN1_PIN,
                  MOTOR_BIN2_PORT, MOTOR_BIN2_PIN, right_dir);
}

static void brake_both(void)
{
    /* 短路制动：IN1=IN2=1。此时 TB6612 把电机两端短接到地，PWM 不参与，
     * 这里把占空比归零只是避免恢复驱动时残留上一次的比较值。 */
    write_directions(MOTOR_DIR_BRAKE, MOTOR_DIR_BRAKE);
    TIM_SetCompare1(MOTOR_TIM, 0U);
    TIM_SetCompare4(MOTOR_TIM, 0U);
}

static void apply_pwm(int16_t left, int16_t right)
{
    uint32_t battery_mv = battery_read_millivolts();
    int32_t limit = (battery_mv >= MOTOR_LIMIT_RESTORE_MV) ?
                    MOTOR_FULL_DUTY_LIMIT : MOTOR_FULL_BATTERY_DUTY_LIMIT;

    if (left > limit) { left = (int16_t)limit; }
    if (left < -limit) { left = (int16_t)(-limit); }
    if (right > limit) { right = (int16_t)limit; }
    if (right < -limit) { right = (int16_t)(-limit); }

    TIM_SetCompare1(MOTOR_TIM, (uint16_t)((left < 0 ? -left : left) *
                                          (MOTOR_PWM_ARR + 1) / 1000));
    TIM_SetCompare4(MOTOR_TIM, (uint16_t)((right < 0 ? -right : right) *
                                          (MOTOR_PWM_ARR + 1) / 1000));
    g_left = left;
    g_right = right;
}

void motor_init(void)
{
#if ENABLE_MOTION_DRIVE
    GPIO_InitTypeDef gpio;
    TIM_TimeBaseInitTypeDef timer;
    TIM_OCInitTypeDef oc;

    if (g_initialized) {

#include "motion.h"
#include "battery.h"
#include "board.h"
#include "board_config.h"

#include "motor.h"
#include "timebase.h"

static uint8_t g_initialized;
static int32_t g_integral_left;
static int32_t g_integral_right;
static int32_t g_last_left;
static int32_t g_last_right;

static int32_t clamp32(int32_t value, int32_t limit)
{
    if (value > limit) { return limit; }
    if (value < -limit) { return -limit; }
    return value;
}

static int16_t pi_update(int32_t target_pps, int32_t actual_pps,
                         int32_t *integral, int32_t *last_actual)
{
    int32_t error = target_pps - actual_pps;
    int32_t derivative;
    int32_t output;

    *integral = clamp32(*integral + error, MOTION_INTEGRAL_LIMIT);
    derivative = actual_pps - *last_actual;
    *last_actual = actual_pps;
    output = (MOTION_PI_KP_NUM * error) / MOTION_PI_KP_DEN +
             (MOTION_PI_KI_NUM * (*integral)) / MOTION_PI_KI_DEN -
             derivative / 2;
    return (int16_t)clamp32(output, 1000);
}

static int32_t target_speed(int32_t target, int32_t done, uint32_t elapsed_ms)
{
    int32_t remaining = target - done;
    uint32_t magnitude = remaining < 0 ? (uint32_t)(-remaining) : (uint32_t)remaining;
    int32_t speed = MOTION_MAX_SPEED_PPS;

    if (magnitude < (uint32_t)MOTION_DECEL_PULSES) {
        speed = MOTION_MIN_SPEED_PPS;
    } else if (elapsed_ms < MOTION_RAMP_MS) {
        speed = MOTION_MIN_SPEED_PPS +
                (int32_t)((uint32_t)(MOTION_MAX_SPEED_PPS - MOTION_MIN_SPEED_PPS) *
                          elapsed_ms / MOTION_RAMP_MS);
    }
    return remaining < 0 ? -speed : speed;
}

static uint8_t drive_to(int32_t left_target, int32_t right_target)
{
    uint32_t start = timebase_millis();
    uint32_t last_control = start;
    uint32_t max_time;

    if (battery_is_undervoltage(battery_read_millivolts())) {
        motor_brake();
        return 0U;
    }
    g_integral_left = 0;
    g_integral_right = 0;
    g_last_left = 0;
    g_last_right = 0;
    max_time = (uint32_t)((left_target < 0 ? -left_target : left_target) *
                          1000U / (uint32_t)MOTION_MIN_SPEED_PPS) +
               MOTION_TIMEOUT_EXTRA_MS;

    while (1) {
        uint32_t now = timebase_millis();
        int32_t left_done;
        int32_t right_done;
        int32_t left_error;
        int32_t right_error;

        board_background_task();
        encoder_update();
        left_done = encoder_left_count();
        right_done = encoder_right_count();
        left_error = left_target - left_done;
        right_error = right_target - right_done;

        if ((left_error > -MOTION_POSITION_TOLERANCE && left_error < MOTION_POSITION_TOLERANCE) &&
            (right_error > -MOTION_POSITION_TOLERANCE && right_error < MOTION_POSITION_TOLERANCE)) {
            motor_brake();
            return 1U;
        }
        if (timebase_expired(now, start, max_time)) {
            motor_brake();
            return 0U;
        }
        if ((uint32_t)(now - last_control) >= MOTION_CONTROL_PERIOD_MS) {
            uint32_t elapsed = now - start;
            int32_t left_speed = target_speed(left_target, left_done, elapsed);
            int32_t right_speed = target_speed(right_target, right_done, elapsed);
            int32_t left_actual = (left_done * 50) / (int32_t)(now - start + 1U);
            int32_t right_actual = (right_done * 50) / (int32_t)(now - start + 1U);
            int16_t left_pwm = pi_update(left_speed, left_actual,
                                         &g_integral_left, &g_last_left);
            int16_t right_pwm = pi_update(right_speed, right_actual,
                                          &g_integral_right, &g_last_right);
            motor_set(left_pwm, right_pwm);
            last_control = now;
        }
    }
}

        encoder_init();
